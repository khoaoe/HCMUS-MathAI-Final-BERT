% Slide 7: Multi-Head Attention
\begin{frame}{Multi-Head Attention}
\begin{columns}
\column{0.5\textwidth}
\begin{center}
\begin{tikzpicture}[scale=0.8]
    % Input
    \node[draw, fill=bertblue!20, minimum width=2cm] (input) at (0,0) {Input};
    
    % Q, K, V projections
    \node[draw, fill=bertorange!20] (q) at (-2,1.5) {Q};
    \node[draw, fill=bertorange!20] (k) at (0,1.5) {K};
    \node[draw, fill=bertorange!20] (v) at (2,1.5) {V};
    
    % Attention heads
    \foreach \i in {1,2,3} {
        \node[draw, fill=bertgreen!20, minimum width=1.5cm] (head\i) at ({(\i-2)*2.0},3) {Head \i};
    }
    
    % Concat
    \node[draw, fill=bertblue!20, minimum width=3cm] (concat) at (0,4.5) {Concat};
    
    % Output projection
    \node[draw, fill=bertblue!20, minimum width=2cm] (output) at (0,6) {Output};
    
    % Connections
    \draw[->] (input) -- (q);
    \draw[->] (input) -- (k);
    \draw[->] (input) -- (v);
    
    \foreach \i in {1,2,3} {
        \draw[->] (q) -- (head\i);
        \draw[->] (k) -- (head\i);
        \draw[->] (v) -- (head\i);
        \draw[->] (head\i) -- (concat);
    }
    
    \draw[->] (concat) -- (output);
\end{tikzpicture}
\end{center}

\column{0.5\textwidth}
\begin{block}{Ý nghĩa các Heads}
\begin{itemize}
    \item<2-> \faEye~Head 1: Quan hệ cú pháp
    \item<3-> \faLink~Head 2: Quan hệ ngữ nghĩa  
    \item<4-> \faGlobe~Head 3: Quan hệ vị trí
    \item<5-> \textbf{...12 heads trong BERT\textsubscript{BASE}}
\end{itemize}
\end{block}

\vspace{0.5em}
\begin{equation*}
\text{Attention}(Q,K,V) = \text{softmax}\left(\frac{QK^T}{\sqrt{d_k}}\right)V
\end{equation*}
\end{columns}
\note{Multi-head attention mechanism - 2 phút}
\end{frame}
